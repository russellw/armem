# Expected verdicts for the C programs, checked by tests/c_programs.rs.
# Each program NAME has NAME.c and the litmus test NAME.litmus, whose
# condition is the bug. Columns: program, optimization level, condition
# reachable on ARM, under TSO, then any further clang flags.
# W_FENCE etc. name an ordering; -DX=__ATOMIC_RELAXED removes it.
# sample=N explores the ARM model on N random paths (for a "yes" only).

seqlock -O2 no no
seqlock -O2 yes no -DW_FENCE=__ATOMIC_RELAXED
seqlock -O2 yes no -DW_SEQ=__ATOMIC_RELAXED
seqlock -O2 yes no -DR_SEQ=__ATOMIC_RELAXED
seqlock -O2 yes no -DR_FENCE=__ATOMIC_RELAXED
seqlock -O0 no no
seqlock -O0 yes no -DW_FENCE=__ATOMIC_RELAXED
seqlock -O0 yes no -DW_SEQ=__ATOMIC_RELAXED
seqlock -O0 yes no -DR_SEQ=__ATOMIC_RELAXED
seqlock -O0 yes no -DR_FENCE=__ATOMIC_RELAXED

spsc -O2 no no
spsc -O2 yes no -DPUB=__ATOMIC_RELAXED
spsc -O2 yes no -DFREE=__ATOMIC_RELAXED
# clang -O2 reuses the earlier read of buf[0], which is wrong everywhere.
spsc -O2 yes yes -DSEE=__ATOMIC_RELAXED
# Ordered on ARM by the control dependency from the load of tail.
spsc -O2 no no -DROOM=__ATOMIC_RELAXED
spsc -O0 no no
spsc -O0 yes no -DPUB=__ATOMIC_RELAXED
spsc -O0 yes no -DFREE=__ATOMIC_RELAXED
spsc -O0 yes no -DSEE=__ATOMIC_RELAXED
spsc -O0 no no -DROOM=__ATOMIC_RELAXED

dcl -O2 no no
dcl -O2 yes no -DPUB=__ATOMIC_RELAXED
# Ordered on ARM by the address dependency from the load of inst.
dcl -O2 no no -DSEE=__ATOMIC_RELAXED
dcl -O2 yes no -DLOCK=__ATOMIC_RELAXED
dcl -O2 yes no -DUNLOCK=__ATOMIC_RELAXED
dcl -O2 no no -DFLAG
dcl -O2 yes no -DFLAG -DPUB=__ATOMIC_RELAXED
# With a flag, only a control dependency orders the read of the value.
dcl -O2 yes no -DFLAG -DSEE=__ATOMIC_RELAXED
dcl -O2 yes no -DFLAG -DLOCK=__ATOMIC_RELAXED
dcl -O2 yes no -DFLAG -DUNLOCK=__ATOMIC_RELAXED
dcl -O0 no no
dcl -O0 yes no -DPUB=__ATOMIC_RELAXED
dcl -O0 no no -DSEE=__ATOMIC_RELAXED
# clang -O0 compiles every compare-and-swap with LDAXR/STLXR.
dcl -O0 no no -DLOCK=__ATOMIC_RELAXED
dcl -O0 yes no -DUNLOCK=__ATOMIC_RELAXED
dcl -O0 no no -DFLAG
dcl -O0 yes no -DFLAG -DPUB=__ATOMIC_RELAXED
dcl -O0 yes no -DFLAG -DSEE=__ATOMIC_RELAXED
dcl -O0 no no -DFLAG -DLOCK=__ATOMIC_RELAXED
dcl -O0 yes no -DFLAG -DUNLOCK=__ATOMIC_RELAXED

refcount -O2 no no
refcount -O2 yes no -DDEC=__ATOMIC_RELAXED
refcount -O2 yes no -DACQ=__ATOMIC_RELAXED
refcount -O0 no no
# clang -O0 compiles every atomic read-modify-write with LDAXR/STLXR.
refcount -O0 no no -DDEC=__ATOMIC_RELAXED
refcount -O0 no no -DACQ=__ATOMIC_RELAXED

treiber -O2 no no
# Too big to explore exhaustively (over 10 minutes); 20 random paths find it.
treiber -O2 yes no -DPUSH=__ATOMIC_RELAXED sample=20
# Ordered on ARM by the address dependency from the load of top.
treiber -O2 no no -DPOP=__ATOMIC_RELAXED
treiber -O0 no no
# clang -O0 compiles every compare-and-swap with LDAXR/STLXR.
treiber -O0 no no -DPUSH=__ATOMIC_RELAXED
